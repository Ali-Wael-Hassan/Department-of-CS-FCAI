cmake_minimum_required(VERSION 3.16)
project(Advanced-Datastrures LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/bin")
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.cpp")
file(GLOB_RECURSE SOURCES_TPP CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/src/*.tpp")
file(GLOB_RECURSE HEADERS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/include/*.h")
file(GLOB_RECURSE HEADERS_HPP CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/include/*.hpp")

if(SOURCES)
    add_executable(${PROJECT_NAME} ${SOURCES} ${HEADERS} ${HEADERS_HPP} ${SOURCES_TPP})
    set_target_properties(${PROJECT_NAME} PROPERTIES OUTPUT_NAME "advanced_datastructures_demo")
    target_include_directories(${PROJECT_NAME} PRIVATE "${CMAKE_SOURCE_DIR}/include" "${CMAKE_SOURCE_DIR}/src")
else()
    add_library(${PROJECT_NAME} INTERFACE)
    target_include_directories(${PROJECT_NAME} INTERFACE "${CMAKE_SOURCE_DIR}/include" "${CMAKE_SOURCE_DIR}/src")
endif()
